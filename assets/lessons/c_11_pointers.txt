int main() {
    int a = 4, b = 9;
    int* big = &a;
    if (b > a) big = &b;
    printf("Bigger: %d\n", *big);
    return 0;
}

@step
~ 3 pointers:1 conditions:0.6 input:0.6 problem_solving:0.8
```
#include <stdio.h>

int main() {
    return 0;
}
```
@input
```
8 3 5
```
@expect
```
8 0 5
```
@solution
```
#include <stdio.h>

int main() {
    int a, b, c;
    scanf("%d %d %d", &a, &b, &c);
    int* smallest = &a;
    if (b < *smallest) smallest = &b;
    if (c < *smallest) smallest = &c;
    *smallest = 0;
    printf("%d %d %d\n", a, b, c);
    return 0;
}
```
