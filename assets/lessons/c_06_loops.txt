    printf("\n");
    return 0;
}

@expect
```
  1   2   3   4
  2   4   6   8
  3   6   9  12
  4   8  12  16
```
@solution
```
#include <stdio.h>

int main() {
    int size = 4;
    for (int row = 1; row <= size; row++) {
        for (int col = 1; col <= size; col++) {
            printf("%3d ", row * col);
        }
        printf("\n");
    }
    return 0;
}
```

#include <stdio.h>

int main() {
    return 0;

@input
```
5
```
@expect
```
5! = 120
```
@solution
```
#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);
    int factorial = 1;
    for (int i = 2; i <= n; i++) {
        factorial *= i;
    }
    printf("%d! = %d\n", n, factorial);
    return 0;
}
```
