#include <iostream>
using namespace std;

int live = 0;

int* getPtrToFive() {
    int* x = new int;
    *x = 5;
    live++;
    return x;
}

void release(int* p) {
    delete p;
    live--;
}

int main() {
    int* p = nullptr;
    for (int i = 0; i < 3; ++i) {
        p = getPtrToFive();
        cout << *p << endl;

}

#include <iostream>
using namespace std;

int live = 0;

int* getPtrToFive() {
    int* x = new int;
    *x = 5;
    live++;
    return x;
}

void release(int* p) {
    delete p;
    live--;
}

int main() {
    int* a = getPtrToFive();
    int* b = getPtrToFive();
    int* c = getPtrToFive();
    cout << *a + *b + *c << endl;
    release(a);
    release(b);
    release(c);
    cout << "live: " << live << endl;
    return 0;
}
