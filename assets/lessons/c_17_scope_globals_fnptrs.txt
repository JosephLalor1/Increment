```
#include <stdio.h>

int Add(int num1, int num2) { return num1 + num2; }
int Subtract(int num1, int num2) { return num1 - num2; }
int Multiply(int num1, int num2) { return num1 * num2; }
int Divide(int num1, int num2) { return num1 / num2; }

int main() {
    int a = 12, b = 4;
    printf("%d\n", Add(a, b));

}
```

```
16 8 48 3
```
@solution
```
#include <stdio.h>

int Add(int num1, int num2) { return num1 + num2; }
int Subtract(int num1, int num2) { return num1 - num2; }
int Multiply(int num1, int num2) { return num1 * num2; }
int Divide(int num1, int num2) { return num1 / num2; }

int main() {
    int a = 12, b = 4;
    int (*ops[4])(int, int) = {Add, Subtract, Multiply, Divide};
    for (int i = 0; i < 4; i++) {
        printf("%d ", ops[i](a, b));
    }
    printf("\n");
    return 0;
}
```
