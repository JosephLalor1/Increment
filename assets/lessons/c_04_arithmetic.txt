```
#include <stdio.h>

int main() {
    return 0;
}
```
@input
```
7 2
```
@expect
```
7 / 2 = 3 r 1
7 / 2 = 3.50
```
@solution
```
#include <stdio.h>

int main() {
    int a, b;
    scanf("%d %d", &a, &b);
    printf("%d / %d = %d r %d\n", a, b, a / b, a % b);
    printf("%d / %d = %.2f\n", a, b, (double)a / b);
    return 0;
}
```
