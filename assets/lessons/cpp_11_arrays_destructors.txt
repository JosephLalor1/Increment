#include <iostream>
using namespace std;

int live = 0;

class Point {
public:
    int x, y;
};

class ManyPoints {
public:
    Point* points;
    ManyPoints() {
        points = new Point[2000];
        live++;
    }
    ~ManyPoints() {
        delete[] points;
        live--;
    }
};

int main() {
    {
        ManyPoints many[10];
    }
    cout << "live: " << live << endl;
    return 0;
}

#include <iostream>
using namespace std;

class Point {
public:
    int x;
    Point(int val) : x(val) {}
    void Print() {
        cout << x << " ";
    }
};

int main() {
    int count;
    cin >> count;
    Point** pointPointers = new Point*[count];
    for (int i = 0; i < count; i++) {
        pointPointers[i] = new Point(i * 10);
    }
    for (int i = 0; i < count; i++) {
        pointPointers[i]->Print();
    }
    cout << endl;
    for (int i = 0; i < count; i++) {
        delete pointPointers[i];
    }
    delete[] pointPointers;
    return 0;
}
