#include <iostream>
using namespace std;

class Temperature {
private:
    double celsius;
public:
    Temperature() : celsius(0) {}
    bool setCelsius(double value) {
        if (value < -273.15) return false;
        celsius = value;
        return true;
    }
    double getCelsius() {
        return celsius;
    }
};

int main() {
    Temperature t;
    cout << t.setCelsius(25) << " " << t.setCelsius(-300) << endl;
    cout << t.getCelsius() << endl;
    return 0;
}
