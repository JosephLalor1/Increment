int* makeSquares(int n) {
    int* squares = (int*)malloc(n * sizeof(int));

#include <stdio.h>
#include <stdlib.h>

int main() {
    int n;
    scanf("%d", &n);
    int* values = (int*)malloc(n * sizeof(int));
    if (values == NULL) {
        return 1;
    }
    for (int i = 0; i < n; i++) {
        scanf("%d", &values[i]);
    }
    for (int i = n - 1; i >= 0; i--) {
        printf("%d ", values[i]);
    }
    printf("\n");
    free(values);
    return 0;
}

#include <stdio.h>
#include <stdlib.h>

int main() {
    int rows, cols;
    scanf("%d %d", &rows, &cols);
    int** grid = (int**)malloc(rows * sizeof(int*));
    for (int row = 0; row < rows; row++) {
        grid[row] = (int*)malloc(cols * sizeof(int));
    }
    int counter = 1;
    for (int row = 0; row < rows; row++) {
        for (int col = 0; col < cols; col++) {
            grid[row][col] = counter++;
        }
    }
    for (int row = 0; row < rows; row++) {
        for (int col = 0; col < cols; col++) {
            printf("%d ", grid[row][col]);
        }
        printf("\n");
    }
    for (int row = 0; row < rows; row++) {
        free(grid[row]);
    }
    free(grid);
    return 0;
}
