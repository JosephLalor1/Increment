#include <iostream>
using namespace std;

class BankAccount {
public:
    int balance;
    BankAccount(int start) : balance(start) {}
    void deposit(int amount) {
        balance += amount;
    }
    void withdraw(int amount) {
        if (amount <= balance) balance -= amount;
    }
    int getBalance() {
        return balance;
    }
};

int main() {
    BankAccount account(100);
    account.deposit(50);
    account.withdraw(30);
    account.withdraw(500);
    cout << "Balance: " << account.getBalance() << endl;
    return 0;
}
