# Q1: Leap years

A leap year is divisible by 4, but not by 100 unless it is also divisible by 400. So 2016 is a leap year, 1700 is not (divisible by 100), and 1600 is (divisible by 400).

```
bool isLeapYear(int year) {
    if (year % 400 == 0) return true;
    if (year % 100 == 0) return false;
    return year % 4 == 0;
}
```

Checking the special cases first (400, then 100) keeps the logic simple. bool, true and false come from #include <stdbool.h>.

# Q2: Reversing digits and palindromes

To reverse a number, repeatedly take its last digit with % 10, add it to the end of the result, and remove it with / 10:

```
int Reversed(int testNumber) {
    int reversed = 0;
    while (testNumber > 0) {
        int digit = testNumber % 10;      // last digit
        reversed = reversed * 10 + digit; // append it
        testNumber = testNumber / 10;     // drop it
    }
    return reversed;
}
```

For 123: the digits come off as 3, 2, 1 and reversed grows 3, 32, 321. A palindrome reads the same both ways, so it equals its own reverse:

```
bool isAPalindrome(int testNumber) {
    return testNumber == Reversed(testNumber);
}
```

# Q3: Prime numbers

A prime is only divisible by itself and 1. You only need to try divisors up to the square root: if n had a larger factor, it would also have a matching smaller one.

```
bool isAPrimeNumber(int n) {
    if (n < 2) return false;
    for (int i = 2; i <= n / i; i++) {
        if (n % i == 0) return false;
    }
    return true;
}
```

> The condition is i <= n / i rather than i * i <= n. For a large n such as 2147483647, i * i would overflow an int before the loop ends.

# Q4: Binary to decimal

The number 110 is written in decimal digits but means binary. Each digit from the right is worth 1, 2, 4, 8 and so on. Take digits off the right with % 10, and double the place value each time, without using the shift operator:

```
int convertBinarytoDecimal(int binaryNumber) {
    int decimal = 0;
    int placeValue = 1;
    while (binaryNumber > 0) {
        int digit = binaryNumber % 10;
        decimal += digit * placeValue;
        placeValue *= 2;
        binaryNumber /= 10;
    }
    return decimal;
}
```

110 becomes 0*1 + 1*2 + 1*4 = 6.

# Putting it together

```
#include <stdio.h>
#include <stdbool.h>

bool isLeapYear(int year) {
    if (year % 400 == 0) return true;
    if (year % 100 == 0) return false;
    return year % 4 == 0;
}

int Reversed(int n) {
    int reversed = 0;
    while (n > 0) {
        reversed = reversed * 10 + n % 10;
        n /= 10;
    }
    return reversed;
}

bool isAPalindrome(int n) {
    return n == Reversed(n);
}

bool isAPrimeNumber(int n) {
    if (n < 2) return false;
    for (int i = 2; i <= n / i; i++) {
        if (n % i == 0) return false;
    }
    return true;
}

int convertBinarytoDecimal(int b) {
    int decimal = 0, placeValue = 1;
    while (b > 0) {
        decimal += (b % 10) * placeValue;
        placeValue *= 2;
        b /= 10;
    }
    return decimal;
}

int main() {
    printf("1700 leap? %d\n", isLeapYear(1700));
    printf("1600 leap? %d\n", isLeapYear(1600));
    printf("Reversed(123) = %d\n", Reversed(123));
    printf("121 palindrome? %d\n", isAPalindrome(121));
    printf("1213 palindrome? %d\n", isAPalindrome(1213));
    printf("2147483647 prime? %d\n", isAPrimeNumber(2147483647));
    printf("110 in decimal = %d\n", convertBinarytoDecimal(110));
    return 0;
}
```

# Q8 and Q9: Searching arrays

find walks the array and returns the index as soon as it sees the value. find2ndLargest keeps track of the largest and second largest seen so far.

```
#include <stdio.h>

int find(int size, int arr[], int toFind) {
    for (int i = 0; i < size; i++) {
        if (arr[i] == toFind) return i;
    }
    return -1;  // not found
}

int find2ndLargest(int size, int arr[]) {
    int largest = -1, second = -1;
    for (int i = 0; i < size; i++) {
        if (arr[i] > largest) {
            second = largest;   // old largest moves down
            largest = arr[i];
        } else if (arr[i] > second && arr[i] != largest) {
            second = arr[i];
        }
    }
    return second;
}

int main() {
    int arr[] = {4, 9, 2, 9, 7};
    printf("find 2: %d\n", find(5, arr, 2));
    printf("find 5: %d\n", find(5, arr, 5));
    printf("2nd largest: %d\n", find2ndLargest(5, arr));
    return 0;
}
```

Output:

```
find 2: 2
find 5: -1
2nd largest: 7
```

> The arr[i] != largest check matters: with two 9s in the array, the second largest is 7, not 9.

#include <stdio.h>

void reverse(int size, int arr[]) {
    for (int i = 0; i < size / 2; i++) {
        int temp = arr[i];
        arr[i] = arr[size - 1 - i];
        arr[size - 1 - i] = temp;
    }
}

void printArray(int size, int arr[]) {
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);

int rotateLeft(int size, int arr[]) {

#include <stdio.h>
#include <stdbool.h>

bool twoMovies(int flightLength, int movieLengths[], int size) {
    for (int i = 0; i < size; i++) {
        for (int j = i + 1; j < size; j++) {  // j > i: two different movies

@solution
```
#include <stdio.h>
#include <stdbool.h>

int wordCounter(int size, char characters[]) {
    int count = 0;
    bool inWord = false;
    for (int i = 0; i < size; i++) {
        char ch = characters[i];
        bool letter = (ch >= 'A' && ch <= 'Z') || (ch >= 'a' && ch <= 'z');
        if (letter && !inWord) count++;  // a new word starts here
        inWord = letter;
    }
    return count;
}

int main() {
    char a[] = "the quick  brown fox";
    char b[] = "  hello  ";
    char c[] = "";
    char d[] = "a b c";
    printf("%d\n", wordCounter(sizeof(a) - 1, a));
    printf("%d\n", wordCounter(sizeof(b) - 1, b));
    printf("%d\n", wordCounter(sizeof(c) - 1, c));
    printf("%d\n", wordCounter(sizeof(d) - 1, d));
    return 0;
}
```
