#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* nextPtr;
};

void InsertAtEnd(Node** ptrToHeadPtr, int newData) {
    Node* newNode = new Node{newData, nullptr};
    Node** link = ptrToHeadPtr;
    while (*link != nullptr) link = &(*link)->nextPtr;
    *link = newNode;
}

void DeleteWithValue(Node** ptrToHeadPtr, int value) {
    Node* curNodePtr = *ptrToHeadPtr;
    if (curNodePtr != nullptr && curNodePtr->data == value) {
        *ptrToHeadPtr = curNodePtr->nextPtr;
        delete curNodePtr;
        return;
    }
    Node* prev = nullptr;
    while (curNodePtr != nullptr && curNodePtr->data != value) {
        prev = curNodePtr;
        curNodePtr = curNodePtr->nextPtr;
    }
    if (curNodePtr == nullptr) return;
    prev->nextPtr = curNodePtr->nextPtr;
    delete curNodePtr;
}

void PrintList(Node* headPtr) {
    for (Node* cur = headPtr; cur; cur = cur->nextPtr) {
        cout << cur->data << " ";
    }
    cout << endl;
}

int main() {
    Node* headPtr = nullptr;
    InsertAtEnd(&headPtr, 10);
    InsertAtEnd(&headPtr, 20);
    InsertAtEnd(&headPtr, 30);
    DeleteWithValue(&headPtr, 20);
    PrintList(headPtr);
    DeleteWithValue(&headPtr, 10);
    PrintList(headPtr);
    return 0;
}

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* nextPtr;
};

void InsertAtBeginning(Node** ptrToHeadPtr, int newData) {
    Node* newNode = new Node{newData, *ptrToHeadPtr};
    *ptrToHeadPtr = newNode;
}

int main() {
    int count;
    cin >> count;
    Node* headPtr = nullptr;
    for (int i = 0; i < count; i++) {
        int value;
        cin >> value;
        InsertAtBeginning(&headPtr, value);
    }
    while (headPtr != nullptr) {
        cout << headPtr->data << " ";
        Node* next = headPtr->nextPtr;
        delete headPtr;
        headPtr = next;
    }
    cout << endl;
    return 0;
}
