#include <stdio.h>

void reverse_array(int* arr, int size) {
    int* left = arr;
    int* right = arr + size - 1;
    while (left < right) {
        int temp = *left;
        *left = *right;
        *right = temp;
        left++;
        right--;
    }
}

int main() {
    int arr[5] = {1, 2, 3, 4, 5};
    reverse_array(arr, 5);
    for (int i = 0; i < 5; i++) {
        printf("%d ", *(arr + i));
    }
    printf("\n");
    return 0;
}

    int values[5];
    for (int* p = values; p < values + 5; p++) {
        scanf("%d", p);
    }
    for (int* p = values + 4; p >= values; p--) {
