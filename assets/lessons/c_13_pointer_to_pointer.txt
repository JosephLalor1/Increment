```

```

```
@solution
```
#include <stdio.h>

void pickLarger(int** result, int* a, int* b) {
    if (*a > *b) *result = a;
    else *result = b;
}

int main() {
    int x = 4, y = 9;
    int* best = &x;
    pickLarger(&best, &x, &y);
    *best = 0;
    printf("x = %d, y = %d\n", x, y);
    return 0;
}
```

```
#include <stdio.h>

int main() {
    int i, j;
    scanf("%d %d", &i, &j);
    int* ptr = &i;
    int** pptr = &ptr;
    **pptr = **pptr * 2;
    *pptr = &j;
    **pptr = **pptr + 1;
    printf("%d %d\n", i, j);
    return 0;
}
