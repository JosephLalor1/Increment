#include <iostream>
#include <cstring>
using namespace std;

class MITStudent {
public:
    int studentID;
    char* name;
    MITStudent() {
        studentID = 0;
        name = NULL;
    }
    MITStudent(MITStudent& o) {
        studentID = o.studentID;
        name = new char[strlen(o.name) + 1];
        strcpy(name, o.name);
    }
};

int main() {
    MITStudent student1;
    student1.studentID = 98;
    char n[] = "foo";
    student1.name = n;
    MITStudent student2 = student1;
    student2.name[0] = 'b';
    cout << student1.name << " " << student2.name << endl;
    delete[] student2.name;
    return 0;
}

#include <iostream>
using namespace std;

class IntArray {
public:
    int size;
    int* data;
    IntArray(int n) : size(n), data(new int[n]) {
        for (int i = 0; i < n; i++) data[i] = i + 1;
    }
    IntArray(const IntArray& o) : size(o.size) {
        data = new int[size];
        for (int i = 0; i < size; i++) data[i] = o.data[i];
    }
    ~IntArray() {
        delete[] data;
    }
    void print() {
        for (int i = 0; i < size; i++) cout << data[i] << " ";
        cout << endl;
    }
};

int main() {
    IntArray original(3);
    IntArray copy = original;
    copy.data[0] = 9;
    original.print();
    copy.print();
    return 0;
}
