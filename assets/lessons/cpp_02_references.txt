#include <iostream>
using namespace std;

void minMax(int arr[], int size, int& min, int& max) {
    min = arr[0];
    max = arr[0];
    for (int i = 1; i < size; i++) {
        if (arr[i] < min) min = arr[i];
        if (arr[i] > max) max = arr[i];
    }
}

int main() {
    int values[] = {45, 12, 89, 34, 67};
    int min = 0, max = 0;
    minMax(values, 5, min, max);
    cout << "min = " << min << ", max = " << max << endl;
    return 0;
}

#include <iostream>
using namespace std;

bool insertElement(int& size, int& count, int arr[], int element,
                   int index) {
    if (count >= size || index < 0 || index > count) return false;
    for (int i = count; i > index; i--) {
        arr[i] = arr[i - 1];
    }
    arr[index] = element;
    count++;
    return true;
}

int main() {
    int arr[6] = {1, 2, 4, 5};
    int size = 6, count = 4;
    insertElement(size, count, arr, 3, 2);
    for (int i = 0; i < count; i++) {
        cout << arr[i] << " ";

    cout << endl << "count = " << count << endl;

#include <iostream>
using namespace std;

bool deleteElement(int& size, int& count, int arr[], int index) {
    if (count > size || index < 0 || index >= count) return false;
    for (int i = index; i < count - 1; i++) {
        arr[i] = arr[i + 1];
    }
    count--;
    return true;
}

int main() {
    int arr[6] = {1, 2, 3, 4, 5};
    int size = 6, count = 5;
    deleteElement(size, count, arr, 1);
    for (int i = 0; i < count; i++) {
        cout << arr[i] << " ";
    }
    cout << endl << "count = " << count << endl;
    return 0;
