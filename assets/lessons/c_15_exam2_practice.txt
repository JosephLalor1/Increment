#include <stdio.h>

int sum_array(int* arr, int size) {
    int total = 0;
    for (int* p = arr; p < arr + size; p++) {
        total += *p;
    }
    return total;
}

int find_max(int* arr, int size) {
    int max = *arr;
    for (int* p = arr + 1; p < arr + size; p++) {
        if (*p > max) max = *p;
    }
    return max;
}

#include <stdio.h>

int count_vowels(char* str) {
    int count = 0;
    for (char* p = str; *p != '\0'; p++) {
        char c = *p;
        if (c >= 'A' && c <= 'Z') c = c + 32;
        if (c == 'a' || c == 'e' || c == 'i') count++;
        if (c == 'o' || c == 'u') count++;
    }
    return count;
}

int main() {
    char str[] = "OpenAI";
    printf("vowels = %d\n", count_vowels(str));
    return 0;
}

#include <stdio.h>

int compare_strings(char* str1, char* str2) {
    while (*str1 != '\0' && *str1 == *str2) {
        str1++;
        str2++;
    }
    return *str1 - *str2;
}

int main() {

#include <stdio.h>

void swap_ints(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void reverse_array(int* arr, int size) {
    for (int* left = arr, *right = arr + size - 1; left < right;
         left++, right--) {
        swap_ints(left, right);
    }
}

int main() {
    int arr[5] = {1, 2, 3, 4, 5};
    reverse_array(arr, 5);
    for (int i = 0; i < 5; i++) {
        printf("%d ", *(arr + i));
    }
    printf("\n");
    return 0;
}
