@solution
```
#include <stdio.h>

int main() {
    char word[] = "banana";
    char target = 'a';
    int count = 0;
    for (int i = 0; word[i] != '\0'; i++) {
        if (word[i] == target) {
            count++;
        }
    }
    printf("%c: %d\n", target, count);
    return 0;
}
```

```
#include <stdio.h>

int main() {
    char name[20] = "Alice";
    int len = 0;
    while (name[len] != '\0') len++;
    for (int i = len - 1; i >= 0; i--) {
        printf("%c", name[i]);
    }
    printf("\n");
    return 0;
}
```
