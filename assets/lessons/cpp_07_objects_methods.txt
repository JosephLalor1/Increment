#include <iostream>
#include <cmath>
using namespace std;

class Point {
public:
    double x, y;
};

class Vector {
public:
    Point start, end;
    double length() {
        double dx = end.x - start.x;
        double dy = end.y - start.y;
        return sqrt(dx * dx + dy * dy);
    }
};

int main() {
    Vector vec;
    vec.start.x = 1.0;
    vec.start.y = 1.0;
    vec.end.x = 4.0;
    vec.end.y = 5.0;
    cout << "length = " << vec.length() << endl;
    return 0;
}

#include <iostream>
using namespace std;

class Rectangle {
public:
    double width, height;
    double area() {
        return width * height;
    }
    void scale(double factor) {
        width *= factor;
        height *= factor;
    }
};

int main() {
    Rectangle r;
    r.width = 3.0;
    r.height = 4.5;
    cout << "area = " << r.area() << endl;
    r.scale(2.0);
    cout << "area = " << r.area() << endl;
    return 0;
}
