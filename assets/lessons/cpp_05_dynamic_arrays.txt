#include <iostream>
using namespace std;

int main() {
    int size = 3;
    int* arr = new int[size];
    for (int i = 0; i < size; ++i) {
        arr[i] = (i + 1) * 10;
    }
    int newSize = 5;
    int* bigger = new int[newSize];
    for (int i = 0; i < newSize; ++i) {
        bigger[i] = i < size ? arr[i] : 0;
    }
    delete[] arr;
    arr = bigger;
    size = newSize;
    for (int i = 0; i < size; ++i) {
        cout << arr[i] << " ";
    }
    cout << endl;
    delete[] arr;
    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int rows = 3;
    int cols = 4;
    int** arr2D = new int*[rows];
    for (int row = 0; row < rows; row++) {
        arr2D[row] = new int[cols];
    }
    int counter = 1;
    for (int row = 0; row < rows; row++) {
        for (int col = 0; col < cols; col++) {
            arr2D[row][col] = counter++;
        }
    }
    for (int row = 0; row < rows; row++) {
        for (int col = 0; col < cols; col++) {
            cout << arr2D[row][col] << " ";
        }
        cout << endl;
    }
    for (int row = 0; row < rows; row++) {
        delete[] arr2D[row];
    }
    delete[] arr2D;
    return 0;
}

int main() {
    int numItems;
    cin >> numItems;
    int* arr = new int[numItems];
    int total = 0;
    for (int i = 0; i < numItems; ++i) {
        cin >> arr[i];
        total += arr[i];
    }
    cout << "Average: " << (double)total / numItems << endl;
    delete[] arr;
